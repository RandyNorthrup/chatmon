"""Write the activation study's workflow, with its scripts embedded (the probe branch holds the workflow file alone)."""

import sys
import textwrap
from pathlib import Path

HERE = Path(__file__).parent


def block(name: str) -> str:
    return textwrap.indent((HERE / name).read_text(encoding="utf-8"), " " * 10)


TEXT = """\
name: Probe screen readers

# A throwaway probe before v0.10.0, on a branch of its own that holds only this file and is deleted afterwards. This
# round: which way brings Chatmon forward when `chatmon ask` comes from a script or launcher on this macOS (cooperative
# activation, macOS 14 and later), for Chatmon run from source and for the installed Chatmon.app, built and installed
# here as the packages workflow does. No screen reader runs. Chatmon's code is 65ed989's, unchanged.
on:
  push:
    branches: [probe/screen-readers]

permissions:
  contents: read

env:
  PIP_DISABLE_PIP_VERSION_CHECK: "1"

jobs:
  activation:
    runs-on: macos-latest
    timeout-minutes: 45
    steps:
      - uses: actions/checkout@3d3c42e5aac5ba805825da76410c181273ba90b1 # v7.0.1
        with:
          persist-credentials: false
      - uses: actions/setup-python@6d4fd22b20465ccc0f7354476588e8df7c9ca0e0 # v7.1.0
        with:
          python-version: "3.14"
      - name: The runner's permissions to drive System Events (guidepup; VoiceOver isn't started)
        uses: guidepup/setup-action@acf1a64cc26292ebe0db6a28d1341bd1bae93c65 # 0.21.0
      - name: Date the build by its commit
        run: echo "SOURCE_DATE_EPOCH=$(git log -1 --format=%ct)" >> "$GITHUB_ENV"
      - name: What the installers are built with, hash-locked
        run: |
          python -m pip install --require-hashes -r packaging/requirements-build.txt
          python -m pip install --require-hashes --no-build-isolation -r packaging/requirements.txt
      - run: python packaging/build.py macos
      - name: Install the .pkg, as a person does
        run: |
          sudo installer -pkg dist/Chatmon-*.pkg -target /
          ls -la /Applications/Chatmon.app/Contents/MacOS
          ls -la /usr/local/bin/chatmon || echo "no /usr/local/bin/chatmon: the .pkg adds it only where /usr/local/bin is root's alone"
      - name: The study's scripts, and Chatmon's settings
        run: |
          mkdir -p "$RUNNER_TEMP/probe" "$RUNNER_TEMP/out"
          cat > "$RUNNER_TEMP/probe/run_chatmon.py" <<'PY'
{shim}          PY
          cat > "$RUNNER_TEMP/probe/activate_try.py" <<'PY'
{try_}          PY
          cat > "$RUNNER_TEMP/probe/study_activation.py" <<'PY'
{study}          PY
          cat > "$RUNNER_TEMP/probe/front_app.py" <<'PY'
{front}          PY
          conf="$HOME/Library/Application Support/chatmon"
          mkdir -p "$conf"
          printf '%s\\n' 'name = "Chatmon"' '[updates]' 'check = false' '[breaks]' 'remind = false' '[heads_up]' 'updates = false' > "$conf/config.toml"
          shasum -a 256 "$RUNNER_TEMP/probe/"*.py | tee "$RUNNER_TEMP/out/scripts.sha256"
      - name: Study
        run: python "$RUNNER_TEMP/probe/study_activation.py" "$RUNNER_TEMP/out" "$GITHUB_WORKSPACE" "$RUNNER_TEMP/probe/run_chatmon.py" "$RUNNER_TEMP/probe/activate_try.py" "$RUNNER_TEMP/probe/front_app.py"
      - name: What ran
        if: ${{{{ always() }}}}
        run: |
          {{ sw_vers; uname -a; python --version; python -m pip freeze; git -C "$GITHUB_WORKSPACE" rev-parse HEAD; }} > "$RUNNER_TEMP/out/environment.txt" 2>&1
          cp "$HOME/Library/Application Support/chatmon/state/chatmon.log" "$RUNNER_TEMP/out/" || true
      - if: ${{{{ always() }}}}
        uses: actions/upload-artifact@cf430e030ddbb5b0abf93d22962f4752f3646cd9 # v7.0.2
        with:
          name: activation-study
          path: ${{{{ runner.temp }}}}/out
          if-no-files-found: error
          retention-days: 30
"""

Path(sys.argv[1]).write_text(TEXT.format(shim=block("run_chatmon.py"), try_=block("activate_try.py"), study=block("study_activation.py"), front=block("front_app.py")),
                             encoding="utf-8", newline="\n")
print("wrote", sys.argv[1])
